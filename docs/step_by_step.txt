FROM apache/spark:3.5.1-python3
USER root
RUN pip install --no-cache-dir jupyterlab  notebook  delta-spark==3.1.0  boto3 --quiet

ADD https://repo1.maven.org/maven2/org/apache/hadoop/hadoop-aws/3.3.4/hadoop-aws-3.3.4.jar /opt/spark/jars/
ADD https://repo1.maven.org/maven2/com/amazonaws/aws-java-sdk-bundle/1.12.262/aws-java-sdk-bundle-1.12.262.jar /opt/spark/jars/


RUN mkdir -p /opt/notebooks
WORKDIR /opt/notebooks
EXPOSE 8888
CMD ["jupyter", "lab", "--ip=0.0.0.0", "--port=8888", "--no-browser", "--NotebookApp.token=", "--NotebookApp.password=", "--allow-root"]
'@ -Encoding utf8

Step 3: Create .yaml file
cd C:\src\rltm_bi_pltfrm

New-Item -Path compose.phase1.yaml -ItemType File -Force

Set-Content -Path compose.phase1.yaml -Value @'
# Phase 1 compose — core lakehouse
services:
  minio:
    image: minio/minio:latest
    container_name: minio
    command: server /data --console-address ":9001"
    environment:
      MINIO_ROOT_USER: ${MINIO_ROOT_USER}
      MINIO_ROOT_PASSWORD: ${MINIO_ROOT_PASSWORD}
    ports:
      - "9000:9000"
      - "9001:9001"
    volumes:
      - ./minio_data:/data

  postgres-warehouse:
    image: postgres:16
    container_name: postgres-warehouse
    environment:
      POSTGRES_USER: ${POSTGRES_USER}
      POSTGRES_PASSWORD: ${POSTGRES_PASSWORD}
      POSTGRES_DB: ${POSTGRES_DB}
    ports:
      - "5432:5432"
    volumes:
      - postgres_warehouse_data:/var/lib/postgresql/data
    healthcheck:
      test: ["CMD-SHELL", "pg_isready -U ${POSTGRES_USER} -d ${POSTGRES_DB}"]
      interval: 15s
      timeout: 5s
      retries: 10

  pgadmin:
    image: dpage/pgadmin4:latest
    container_name: pgadmin
    environment:
      PGADMIN_DEFAULT_EMAIL: admin@example.com
      PGADMIN_DEFAULT_PASSWORD: admin123
    ports:
      - "5050:80"
    depends_on:
      - postgres-warehouse

  spark-master:
    image: apache/spark:3.5.1-python3
    container_name: spark-master
    command: >
      /opt/spark/bin/spark-class org.apache.spark.deploy.master.Master
    environment:
      SPARK_NO_DAEMONIZE: "true"
    ports:
      - "7077:7077"
      - "8080:8080"
    volumes:
      - ./jars:/opt/spark/jars-extra
      - ./jobs:/opt/spark/apps

  spark-worker:
    image: apache/spark:3.5.1-python3
    container_name: spark-worker
    depends_on:
      - spark-master
    command: >
      /opt/spark/bin/spark-class org.apache.spark.deploy.worker.Worker spark://spark-master:7077
    environment:
      SPARK_NO_DAEMONIZE: "true"
      SPARK_WORKER_CORES: "2"
      SPARK_WORKER_MEMORY: "2G"
    ports:
      - "8081:8081"
    volumes:
      - ./jars:/opt/spark/jars-extra
      - ./jobs:/opt/spark/apps

  jupyter:
    build:
      context: .
      dockerfile: docker/jupyter/Dockerfile
    container_name: jupyter
    depends_on:
      - spark-master
      - minio
      - postgres-warehouse
    environment:
      SPARK_MASTER: spark://spark-master:7077
      AWS_ACCESS_KEY_ID: ${MINIO_ROOT_USER}
      AWS_SECRET_ACCESS_KEY: ${MINIO_ROOT_PASSWORD}
      AWS_REGION: us-east-1
    ports:
      - "8888:8888"
    volumes:
      - ./:/workspace/rltm_bi_pltfrm
      - ./jars:/opt/spark/jars-extra
	  

volumes:
  postgres_warehouse_data:
'@ -Encoding utf8
